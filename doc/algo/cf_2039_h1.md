    def performSwaps(skips):
        o = []
        skips = skips[::-1]
        i = 0

        while i+1 < n:
            if skips:
                if i+1 == skips[-1]:
                    i += 2
                    skips.pop()
                    o.append("DDRR")
                    continue
        
            i += 1
            o.append("DR")

        return o

            #it's sorted now
